#To verify postgresql (gives "9760    0    postgresql-18" when started and gives not output when stopped)

#To stop postgresql

   ./asi-venv/bin/python -c "import sys; print(sys.executable)"

   ./asi-venv/bin/python -m ipykernel install --user --name asi-venv --display-name "Python (asi-venv)"
